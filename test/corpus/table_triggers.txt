================================================================================
table triggers OnInsert, OnModify, OnDelete and OnRename
================================================================================

table 50100 "Tour Member"
{
    fields
    {
        field(1; "No."; Code[20]) { }
    }

    trigger OnInsert()
    begin
        if Balance = 0 then
            Balance := 100;
    end;

    trigger OnModify()
    begin
        "Last Balance" := xRec.Balance;
    end;

    trigger OnDelete()
    begin
        Error('Cannot delete %1', Rec."No.");
    end;

    trigger OnRename()
    begin
        "Old No." := xRec."No.";
    end;
}

                if Name <> xRec.Name then

                Page.RunModal(0, Rec);
            end;
