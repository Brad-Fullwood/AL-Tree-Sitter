local label used by StrSubstNo and Error

codeunit 50100 "Label Probe"
{
    procedure Greet(Name: Text): Text
    var
        GreetingLbl: Label 'Hello %1, you have %2 items', Comment = '%1 = name, %2 = count';
        EmptyErr: Label 'Name must not be empty.';
    begin
        if Name = '' then
            Error(EmptyErr);
        exit(StrSubstNo(GreetingLbl, Name, 3));
    end;
}
