================================================================================
ReadFrom and SelectToken with a var token
================================================================================

codeunit 50100 "Json Probe"
{
    procedure Read(): Text
    begin
        if not Doc.ReadFrom('{"order":{"no":"SO1"}}') then
            exit('');
        Doc.SelectToken('$.order.no', Token);
        exit(Token.AsValue().AsText());
    end;
}

================================================================================
JSON array access inside a for loop
================================================================================

codeunit 50100 "Json Probe"
{
    procedure Sum(): Decimal
    begin
        Items := Token.AsArray();
        for i := 0 to Items.Count() - 1 do begin
            Items.Get(i, Token);
            Total += Token.AsObject().GetDecimal('qty');
        end;
        exit(Total);
    end;
}

codeunit 50100 "Json Probe"
{
    procedure Write(): Text
    begin
        Customer.Add('name', 'Ann');
        Customer.Add('vip', true);
        Lines.Add(1);
        Customer.Add('lines', Lines);
        Customer.Replace('name', 'Bo');
        Customer.WriteTo(Out);
        exit(Format(Customer.Contains('name')) + Out);
    end;
}
