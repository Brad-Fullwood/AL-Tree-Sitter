{
    SingleInstance = true;
    EventSubscriberInstance = Manual;

    var
        Deleted: Integer;

    [TryFunction]
    procedure TryCount(var Total: Integer)
    begin
        Total := Deleted;
    end;

    [EventSubscriber(ObjectType::Table, Database::"Sales Header", 'OnStamp', '', false, false)]
    local procedure OnStampSub(var Sender: Record "Sales Header"; Times: Integer)
    begin
        Sender.Note := 'stamped ' + Sender."No." + ' ' + Format(Times);
    end;

    [EventSubscriber(ObjectType::Table, Database::Customer, 'OnAfterDeleteEvent', '', false, false)]
    local procedure CountDelete(var Rec: Record Customer; RunTrigger: Boolean)
    begin
        Deleted += 1;
    end;

    [EventSubscriber(ObjectType::Table, 18, 'OnAfterDeleteEvent', '', false, false)]
    local procedure CountDeleteById(var Rec: Record Customer; RunTrigger: Boolean)
    begin
        Deleted += 1;
    end;
}

codeunit 50110 "Codeunit Variables"
{
    procedure Reset(): Integer
    var
        Cu: Codeunit "Header Subscribers";
        OtherCu: Codeunit "Header Subscribers";
        Total: Integer;
    begin
        Cu := OtherCu;
        Clear(Cu);
        BindSubscription(Cu);
        if Cu.TryCount(Total) then
            exit(Total);
        UnbindSubscription(Cu);
    end;
}
