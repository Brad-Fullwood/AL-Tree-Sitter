================================================================================
sign after a comparison, a bracket, a parenthesis and a comma
================================================================================

table 50102 "Sign Ledger"
{
    fields
    {
        field(1; Amount; Decimal) { }
    }
}

codeunit 50102 "Sign Probe"
{
    procedure Compare(X: Integer): Integer
    var
        Rec: Record "Sign Ledger";
    begin
        if X<-1 then
            X := 0;
        if X>-1 then
            X := 0;
        if (X<>-1) and (X=-1) then
            X := 0;
        if X<=-1 then
            X := 0;
        if X in [-5..-2] then
            X := 0;
        Rec.SetRange(Amount, -100, 0);
        exit(-1);
    end;
}
