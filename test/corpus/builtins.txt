codeunit 50100 "Builtin Probe"
{
    procedure Build(): Text
    var
        Builder: TextBuilder;
        Id: Guid;
    begin
        Builder.Append('a');
        Builder.AppendLine('b');
        Id := CreateGuid();
        if IsNullGuid(Id) then
            exit('');
        exit(Builder.ToText() + Format(Id));
    end;
}

codeunit 50100 "Builtin Probe"
{
    procedure Fill()
    var
        Txt: array[3] of Text;
        Grid: array[2, 3] of Integer;
        i: Integer;
    begin
        Txt[1] := 'a';
        for i := 2 to ArrayLen(Txt) do
            Txt[i] := Txt[i - 1] + 'b';
        Grid[1, 2] := 5;
    end;
}

codeunit 50100 "Builtin Probe"
{
    procedure Dates(): Text
    begin
        D := CalcDate('<+1M>', 20240131D);
        D := CalcDate('<CM>', Today());
        if not Evaluate(N, '42') then
            N := 0;
        exit(StrSubstNo('%1 %2', D, N));
    end;
}
