codeunit 50101 "Tour Tests"
{
    [Test]
    procedure DeleteFails()
    begin
        asserterror Member.Delete(true);
        asserterror begin
            Member.Validate(Balance, -1);
            Member.Modify();
        end;
        asserterror Error('Balance %1', 5);
        if GetLastErrorText() <> 'Balance 5' then
            Error('Unexpected error: %1', GetLastErrorText());
    end;
}
