List range methods and the var result form of GetRange

codeunit 50101 "List Probe"
{
    procedure Ranges(var Names: List of [Text]; Other: List of [Text]): Integer
    var
        Part: List of [Text];
    begin
        Names.AddRange(Other);
        Names.AddRange('a', 'b');
        Part := Names.GetRange(1, 2);
        Names.GetRange(2, 1, Part);
        Names.RemoveRange(1, 2);
        Names.Reverse();
        exit(Names.LastIndexOf('a'));
    end;
}

================================================================================
foreach over a List and over Dictionary keys with Get in an if
================================================================================

codeunit 50101 "List Probe"
{
    procedure Loops(Names: List of [Text]; Prices: Dictionary of [Code[20], Decimal]) Total: Decimal
    var
        Name: Text;
        ItemNo: Code[20];
        Value: Decimal;
    begin
        foreach Name in Names do
            Total += StrLen(Name);
        foreach ItemNo in Prices.Keys do
            if Prices.Get(ItemNo, Value) then
                Total += Value;
    end;
}
