codeunit 50104 "Keyword Names More"
{
    procedure Params(Value: Decimal; Code: Code[20]; var Page: Text; Record: Integer): Text
    var
        Enum: Integer;
        Object: Text;
        Label: Text;
        Letter: Char;
    begin
        for Enum := 1 to 3 do
            Value += Enum;
        for Record := 3 downto 1 do
            Value -= Record;
        case Code of
            'A':
                Page := Code;
        end;
        Evaluate(Value, '1.5');
        Label := Page.ToUpper;
        Object := Label.Trim();
        Page[1] := 'x';
        Letter := Code[2];
        exit(LowerCase(Code) + Object);
    end;
}
