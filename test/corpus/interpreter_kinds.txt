================================================================================
if with an empty then, an empty else and both branches
================================================================================

codeunit 50111 "If Shapes"
{
    procedure Run(A: Boolean) Result: Integer
    var
        Flag: Boolean;
    begin
        Flag := true;
        if A then;
        if A then
            Result := 1
        else;
        if A and (Flag <> false) then
            Result := 3
        else
            Result := 4;
    end;
}
