codeunit 50105 "Case Labels"
{
    procedure Classify(X: Integer; A: Integer): Integer
    var
        Y: Integer;
    begin
        case X of
            -1:
                exit(1);
            - 2:
                exit(2);
            -Level::Gold.AsInteger():
                exit(3);
            -5..-3:
                exit(4);
            -10..-6, 10..20:
                exit(5);
            0:
                Y := A -1;
            else
                exit(0);
        end;
        exit(Y);
    end;
}

{
    value(0; Bronze) { }
    value(1; Silver) { }
    value(2; Gold) { }
}

codeunit 50105 "Case Labels"
{
    procedure Letters(C: Text[1]): Integer
    begin
        case C of
            'a'..'f':
                exit(1);
            'g', 'h':
                exit(2);
        end;
    end;

    procedure Levels(L: Enum Level): Integer
    begin
        case L of
            Enum::Level::Bronze, Enum::Level::Silver:
                exit(1);
            Level::Gold:
                exit(2);
        end;
    end;
}

================================================================================
else arms with and without a semicolon

codeunit 50106 "Case Else"
{
    procedure EmptyElse(X: Integer): Integer
    begin
        case X of
            1:
                exit(1);
            else;
        end;
        case X of
            1:
                exit(1)
            else
        end;
        case X of
            1:
                ;
            else
                exit(2);
        end;
        case X of
            2:
                X := 3
            else
                X := 4;
        end;
        exit(X);
    end;
}
